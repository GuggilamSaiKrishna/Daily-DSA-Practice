class Solution {
    public int kthElement(int a[], int b[], int k) {
        // code here
        int n1 = a.length; 
        int n2 = b.length;
        int n3 = n1+n2;
        int c[] = new int[n3];
        int d=0;
        for(int i = 0;i<n1;i++)
        {
            c[d]=a[i];
            d++;
        }
        for(int i = 0;i<n2;i++)
        {
            c[d]=b[i];
            d++;
        }
        Arrays.sort(c);
        return c[k-1];
    }
}

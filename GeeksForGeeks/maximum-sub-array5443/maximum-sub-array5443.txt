class Solution {
    public ArrayList<Integer> findSubarray(int arr[]) {
        ArrayList<Integer> ans = new ArrayList<>();
        ArrayList<Integer> temp = new ArrayList<>();

        for (int i = 0; i < arr.length; i++) {
            if (arr[i] >= 0) {
                temp.add(arr[i]);
            } else {
                if (temp.size() > ans.size()) {
                    ans = new ArrayList<>(temp);
                }
                temp.clear();
            }
        }

        if (temp.size() > ans.size()) {
            ans = new ArrayList<>(temp);
        }

        if (ans.isEmpty()) {
            ans.add(-1);
        }

        return ans;
    }
}

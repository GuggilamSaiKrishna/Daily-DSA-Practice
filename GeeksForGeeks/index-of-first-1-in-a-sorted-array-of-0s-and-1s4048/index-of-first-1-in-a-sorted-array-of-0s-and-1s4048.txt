class Solution {
    public int firstIndex(int arr[]) {
        // code here
        int count=0;
        for(int i = 0; i<arr.length ; i++)
        {
            if(arr[i]==0)
            {
                count++;
            }
        }
        if(count == arr.length)
        {
            return -1;
        }
        return count;
    }
}

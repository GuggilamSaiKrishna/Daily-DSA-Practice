
class Solution {
    boolean pythagoreanTriplet(int[] arr) {
        int n = arr.length;

        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                for (int k = j + 1; k < n; k++) {
                    long a = arr[i];
                    long b = arr[j];
                    long c = arr[k];

                    if (a * a + b * b == c * c ||
                        a * a + c * c == b * b ||
                        b * b + c * c == a * a) {
                        return true;
                    }
                }
            }
        }

        return false;
    }
}

class Solution {
    ArrayList<Integer> removeDuplicates(int[] arr) {
        ArrayList<Integer> arr1 = new ArrayList<>();

        int n = arr.length;

        for (int i = 0; i < n - 1; i++) {
            if (arr[i] != arr[i + 1]) {
                arr1.add(arr[i]);
            }
        }

        if (n > 0) {
            arr1.add(arr[n - 1]);
        }

        return arr1;
    }
}

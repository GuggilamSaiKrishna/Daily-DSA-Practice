    public static ArrayList<ArrayList<Integer>> getPairs(int[] arr) {

        ArrayList<ArrayList<Integer>> ans = new ArrayList<>();

        Arrays.sort(arr);

        int i = 0;
        int j = arr.length - 1;

        while (i < j) {

            int sum = arr[i] + arr[j];

            if (sum == 0) {

                ArrayList<Integer> pair = new ArrayList<>();
                pair.add(arr[i]);
                pair.add(arr[j]);

                ans.add(pair);

                int left = arr[i];
                int right = arr[j];

                while (i < j && arr[i] == left) {
                    i++;
                }

                while (i < j && arr[j] == right) {
                    j--;
                }
            }
            else if (sum < 0) {
                i++;
            }

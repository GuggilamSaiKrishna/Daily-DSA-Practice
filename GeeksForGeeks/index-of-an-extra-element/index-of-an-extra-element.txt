class Solution {
    public int findExtra(int a[], int b[]) {

        int i = 0;
        int j = 0;

        while (i < a.length && j < b.length) {

            if (a[i] == b[j]) {
                i++;
                j++;
            } 
            else {
                return i;
            }
        }

        return i;
    }
}

class Solution {
    public boolean findPair(int[] arr, int x) {
        for (int i = 0; i < arr.length; i++) {
            for (int j = i + 1; j < arr.length; j++) {
                // Check if the absolute difference between the two elements equals x
                if (Math.abs(arr[i] - arr[j]) == x) {
                    return true;
                }
            }
        }
        return false;
    }
}

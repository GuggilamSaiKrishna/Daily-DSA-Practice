class Solution {
    public long totalFine(int date, int car[], int fine[]) {
        long tfine = 0;

        for (int i = 0; i < car.length; i++) {
            if (date % 2 == 0 && car[i] % 2 != 0) {
                tfine += fine[i];
            } 
            else if (date % 2 != 0 && car[i] % 2 == 0) {
                tfine += fine[i];
            }
        }

        return tfine;
    }
}

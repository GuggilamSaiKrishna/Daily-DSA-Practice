class Solution {
    public void segregateElements(int[] arr) {
        // code herearr.length;
        int n = arr.length;
        int [] arr1 = new int[n];
        int [] arr2 = new int[n];
        int k=0,j=0,m=0;
        for(int i = 0; i<n ; i++)
        {
            if(arr[i]>= 0)
            {
                arr1[k]=arr[i];
                k++;
            }
            else{
                arr2[j]=arr[i];
                j++;
            }
        }
        for(int i = 0 ; i<k ; i++)
        {
            arr[m] = arr1[i];
            m++;
        }
        for(int i = 0 ; i<j ; i++)
        {
            arr[m] = arr2[i];
            m++;
        }
    }
}

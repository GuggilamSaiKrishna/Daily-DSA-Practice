
import java.util.*;

class Solution {
    boolean pythagoreanTriplet(int[] arr) {
        HashSet<Long> squares = new HashSet<>();
        int n = arr.length;

        for (int x : arr) {
            long v = x;
            squares.add(v * v);
        }

        for (int i = 0; i < n; i++) {
            long a = arr[i];
            long a2 = a * a;

            for (int j = i + 1; j < n; j++) {
                long b = arr[j];
                long sum = a2 + b * b;

                if (squares.contains(sum)) {
                    return true;
                }
            }
        }

        return false;
    }
}

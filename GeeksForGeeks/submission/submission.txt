class Solution {
    int typeOfArr(int arr[]) {
        int n = arr.length;
        int asc = 0, desc = 0;

        for (int i = 0; i < n - 1; i++) {
            if (arr[i] > arr[i + 1]) {
                asc++;
            } else {
                desc++;
            }
        }

        if (asc == 0)
            return 1;

        if (desc == 0)
            return 2;

        if (asc == 1 && arr[n - 1] < arr[0])
            return 4;

        if (desc == 1 && arr[n - 1] > arr[0])
            return 3;

        return 0;
    }
}

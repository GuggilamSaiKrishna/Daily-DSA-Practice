class Solution {
    public int firstSearch(int[] arr, int k) {
        Arrays.sort(arr);
        int c = 0;
        if (k < 1 || k > arr.length) {
            return -1;
        }

        for(int i = 0 ; i<arr.length;i++)
        {
            if(arr[i]==k){
                c = i;
            }
        }
        return c;
    }
}

class Solution {
    void rearrange(ArrayList<Integer> arr) {
        ArrayList<Integer> ans1 = new ArrayList<>();
        ArrayList<Integer> ans2 = new ArrayList<>();

        for (int i = 0; i < arr.size(); i++) {
            if (arr.get(i) >= 0) {
                ans1.add(arr.get(i));
            } else {
                ans2.add(arr.get(i));
            }
        }

        int k = 0, j = 0;

        for (int i = 0; i < arr.size(); i++) {
            if (i % 2 == 0 && k < ans1.size()) {
                arr.set(i, ans1.get(k));
                k++;
            } else if (i % 2 != 0 && j < ans2.size()) {
                arr.set(i, ans2.get(j));
                j++;
            } else if (k < ans1.size()) {
                arr.set(i, ans1.get(k));
                k++;
            } else {
                arr.set(i, ans2.get(j));
                j++;
            }
        }
    }
}

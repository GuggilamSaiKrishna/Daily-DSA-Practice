class Solution {
    public ArrayList<Integer> minAnd2ndMin(int[] arr) {
        // code here
        ArrayList<Integer> result = new ArrayList<>();
        Arrays.sort(arr);
        if(arr.length<2){
            return result;
        }
        else{
        result.add(arr[0]);
        result.add(arr[1]);
        return result;
        }
    }
}

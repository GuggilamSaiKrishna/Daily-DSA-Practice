class Solution {
    public ArrayList<Integer> findIntersection(int[] a, int[] b) {
        ArrayList<Integer> c = new ArrayList<>();
        HashSet<Integer> set = new HashSet<>();

        for (int i = 0; i < b.length; i++) {
            set.add(b[i]);
        }

        for (int i = 0; i < a.length; i++) {
            if (set.contains(a[i])) {
                c.add(a[i]);
                set.remove(a[i]);
            }
        }

        return c;
    }
}

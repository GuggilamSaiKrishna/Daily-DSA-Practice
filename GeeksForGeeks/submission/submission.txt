class Solution {
    ArrayList<Integer> removeDuplicates(int[] arr) {
        ArrayList<Integer> arr1 = new ArrayList<>();

        int n = arr.length;

        for (int i = 0; i < n - 1; i++) {
            if (arr[i] != arr[i + 1]) {
                arr1.add(arr[i]);
            }
        }

      

        return arr1;
    }
}

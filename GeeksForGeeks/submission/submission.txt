class Solution {
    public int visibleBuildings(int[] arr) {
        int n = arr.length;
        int count = 0;
        int max = Integer.MIN_VALUE;

        for (int i = 0; i < n; i++) {
            if (arr[i] > max) {
                count++;
                max = arr[i];
            }
        }

        return count;
    }
}

class Solution {
    public boolean checkElements(int start, int end, int[] arr) {
        for (int num = start; num <= end; num++) {
            boolean found = false;

            for (int i = 0; i < arr.length; i++) {
                if (arr[i] == num) {
                    found = true;
                    break;
                }
            }

            if (!found) {
                return false;
            }
        }
        return true;
    }
}

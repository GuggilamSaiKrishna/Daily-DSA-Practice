class Solution {
    public ArrayList<Integer> commonElements(int[] a, int[] b, int[] c) {
        // code here
        ArrayList<Integer>arr = new ArrayList<>();
        for(int i = 0;i<a.length;i++)
        {
            for(int j = 0 ;j<b.length;j++)
            {
                for(int k = 0;k<c.length;k++)
                {
                    if(a[i]==b[j] && b[j]==c[k])
                    {
                         if (!arr.contains(a[i])){
                             arr.add(a[i]);
                         }
                        
                    }
                }
            }
        }
        return arr;
    }
}
